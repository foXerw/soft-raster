# 02 — 数学库：四条约定，和一套能核对的 Vec/Mat/BBox

## 这一章要解决什么问题

第 01 章给了你窗口和一块自己能写的像素内存。要画三角形，你得回答"这个顶点在屏幕的哪里"，答案是一串矩阵乘法。

## 最小片段一：Vec3 + Mat4（行主序、余子式求逆、法线几何）

读代码前先记住一件事：`m[row][col]`，`result[r][c] = Σ_k a[r][k]·b[k][c]`，于是 `proj * view * model * p`
是**最右边的先作用**。main 里四组打印分别钉住：求逆往返、乘法不可交换、法线的 inverse-transpose 几何、零向量与近奇异约定。

```cpp
// snippet1.cpp — Vec3 + Mat4（行主序、列向量右乘、余子式求逆）+ 法线几何
// g++ -std=c++20 -Wall -Wextra -ffp-contract=off snippet1.cpp -o snippet1 && ./snippet1
#include <cassert>
#include <cmath>
#include <cstdio>

struct Vec3 { float x{}, y{}, z{}; };
inline Vec3 operator+(const Vec3& a, const Vec3& b) { return {a.x + b.x, a.y + b.y, a.z + b.z}; }
inline Vec3 operator-(const Vec3& a, const Vec3& b) { return {a.x - b.x, a.y - b.y, a.z - b.z}; }
inline Vec3 operator*(const Vec3& a, float s) { return {a.x * s, a.y * s, a.z * s}; }
inline float dot(const Vec3& a, const Vec3& b) { return a.x * b.x + a.y * b.y + a.z * b.z; }
inline Vec3 normalize(const Vec3& a) {
    const float l2 = dot(a, a);
    if (!(l2 > 1e-16f)) return {};   // 零向量与 NaN 都走这条安全分支
    return a * (1.0f / std::sqrt(l2));
}
struct Vec4 { float x{}, y{}, z{}, w{}; };
inline Vec4 makeVec4(const Vec3& v, float w) { return {v.x, v.y, v.z, w}; }
inline Vec3 xyz(const Vec4& v) { return {v.x, v.y, v.z}; }

struct Mat4 {
    float m[4][4]{};   // 行主序：m[row][col]，配列向量右乘 p' = M * p
    static Mat4 identity() { Mat4 r{}; r.m[0][0] = r.m[1][1] = r.m[2][2] = r.m[3][3] = 1; return r; }
    static Mat4 translation(const Vec3& t) {
        Mat4 r = identity(); r.m[0][3] = t.x; r.m[1][3] = t.y; r.m[2][3] = t.z; return r;
    }
    static Mat4 scale(const Vec3& s) {
        Mat4 r{}; r.m[0][0] = s.x; r.m[1][1] = s.y; r.m[2][2] = s.z; r.m[3][3] = 1; return r;
    }
    static Mat4 rotationY(float a) {
        const float c = std::cos(a), s = std::sin(a);
        Mat4 r = identity(); r.m[0][0] = c; r.m[0][2] = s; r.m[2][0] = -s; r.m[2][2] = c; return r;
    }
    Mat4 operator*(const Mat4& b) const {   // result[r][c] = sum_k a[r][k]*b[k][c]
        Mat4 r{};
        for (int row = 0; row < 4; ++row)
            for (int col = 0; col < 4; ++col)
                for (int k = 0; k < 4; ++k) r.m[row][col] += m[row][k] * b.m[k][col];
        return r;
    }
    Vec4 operator*(const Vec4& v) const {
        return {m[0][0] * v.x + m[0][1] * v.y + m[0][2] * v.z + m[0][3] * v.w,
                m[1][0] * v.x + m[1][1] * v.y + m[1][2] * v.z + m[1][3] * v.w,
                m[2][0] * v.x + m[2][1] * v.y + m[2][2] * v.z + m[2][3] * v.w,
                m[3][0] * v.x + m[3][1] * v.y + m[3][2] * v.z + m[3][3] * v.w};
    }
    Mat4 transposed() const {
        Mat4 r{};
        for (int i = 0; i < 4; ++i) for (int j = 0; j < 4; ++j) r.m[j][i] = m[i][j];
        return r;
    }
    Mat4 upper3x3AsMat4() const {
        Mat4 r{};
        for (int i = 0; i < 3; ++i) for (int j = 0; j < 3; ++j) r.m[i][j] = m[i][j];
        r.m[3][3] = 1;
        return r;
    }
};
inline float det3(const float m[3][3]) {
    return m[0][0] * (m[1][1] * m[2][2] - m[1][2] * m[2][1]) -
           m[0][1] * (m[1][0] * m[2][2] - m[1][2] * m[2][0]) +
           m[0][2] * (m[1][0] * m[2][1] - m[1][1] * m[2][0]);
}
inline float cofactor(const float m[4][4], int r, int c) {   // 删第 r 行第 c 列，带 (-1)^(r+c)
    float sub[3][3];
    int si = 0;
    for (int i = 0; i < 4; ++i) {
        if (i == r) continue;
        int sj = 0;
        for (int j = 0; j < 4; ++j) { if (j == c) continue; sub[si][sj++] = m[i][j]; }
        ++si;
    }
    const float d = det3(sub);
    return (((r + c) & 1) != 0) ? -d : d;
}
// A^-1 = adj(A)/det(A)，adj = 余子式矩阵的转置。失败 ⇒ false 且不写 out。
inline bool tryInvert(const Mat4& in, Mat4& out) {
    float cof[4][4];
    for (int r = 0; r < 4; ++r)
        for (int c = 0; c < 4; ++c) cof[r][c] = cofactor(in.m, r, c);
    const float det = in.m[0][0] * cof[0][0] + in.m[0][1] * cof[0][1] +
                      in.m[0][2] * cof[0][2] + in.m[0][3] * cof[0][3];
    float maxAbs = 0;
    for (int i = 0; i < 4; ++i)
        for (int j = 0; j < 4; ++j)
            if (std::fabs(in.m[i][j]) > maxAbs) maxAbs = std::fabs(in.m[i][j]);
    const float eps = 1e-8f * maxAbs * maxAbs * maxAbs;   // 相对量级阈值，不是固定 1e-8
    if (!(std::fabs(det) > eps)) return false;
    const float invDet = 1.0f / det;
    for (int r = 0; r < 4; ++r)
        for (int c = 0; c < 4; ++c) out.m[r][c] = cof[c][r] * invDet;   // 注意这里的转置
    return true;
}

int main() {
    // 1) 求逆往返：平移 + 非均匀缩放 + 旋转，最容易暴露余子式索引错误的组合
    const Mat4 a = Mat4::translation({1.5f, -2.0f, 3.5f}) *
                   Mat4::scale({2.0f, 0.5f, 4.0f}) * Mat4::rotationY(0.7f);
    Mat4 inv{};
    assert(tryInvert(a, inv));
    const Mat4 prod = a * inv, id = Mat4::identity();
    float maxErr = 0;
    for (int r = 0; r < 4; ++r)
        for (int c = 0; c < 4; ++c)
            if (std::fabs(prod.m[r][c] - id.m[r][c]) > maxErr) maxErr = std::fabs(prod.m[r][c] - id.m[r][c]);
    assert(maxErr < 1e-4f);
    std::printf("max |M*M^-1 - I| = %.3e\n", static_cast<double>(maxErr));
    // 2) 不可交换：c=平移(2,3,4)，d=缩放(2,3,4)，看平移列
    const Mat4 c = Mat4::translation({2, 3, 4}), d = Mat4::scale({2, 3, 4});
    const Mat4 cd = c * d, dc = d * c;
    std::printf("(c*d) translation col = %.0f %.0f %.0f\n",
                double(cd.m[0][3]), double(cd.m[1][3]), double(cd.m[2][3]));
    std::printf("(d*c) translation col = %.0f %.0f %.0f\n",
                double(dc.m[0][3]), double(dc.m[1][3]), double(dc.m[2][3]));
    // 3) 法线几何：scale(2,1,1)，n=(1,1,0)/√2，切向 t1=(1,-1,0)/√2、t2=(0,0,1)
    const Mat4 model = Mat4::scale({2, 1, 1});
    const Vec3 n = normalize({1, 1, 0}), t1 = normalize({1, -1, 0}), t2{0, 0, 1};
    assert(std::fabs(dot(n, t1)) < 1e-5f && std::fabs(dot(n, t2)) < 1e-5f);   // 前提：n 垂直于表面
    const Vec3 t1w = xyz(model * makeVec4(t1, 0)), t2w = xyz(model * makeVec4(t2, 0));
    const Vec3 naive = xyz(model * makeVec4(n, 0));   // (2,1,0)/√2
    std::printf("naive: dot(n', M*t1) = %.4f   (unnormalized tangent: %.4f)\n",
                double(dot(naive, t1w)), double(dot(naive, xyz(model * makeVec4({1, -1, 0}, 0)))));
    Mat4 inv3{};
    assert(tryInvert(model.upper3x3AsMat4(), inv3));
    const Vec3 nOut = normalize(xyz(inv3.transposed() * makeVec4(n, 0)));   // 逆转置 = diag(0.5,1,1)
    std::printf("normal matrix: dot(nOut, M*t1) = %.3e, dot(nOut, M*t2) = %.3e\n",
                double(dot(nOut, t1w)), double(dot(nOut, t2w)));
    // 4) 零向量约定与近奇异拒绝
    const Vec3 z = normalize({0, 0, 0});
    assert(z.x == 0 && z.y == 0 && z.z == 0);
    std::printf("normalize(0) = (%g, %g, %g)\n", double(z.x), double(z.y), double(z.z));
    Mat4 unused{};
    std::printf("tryInvert(scale(1e-5,1e-5,1)) = %s\n",
                tryInvert(Mat4::scale({1e-5f, 1e-5f, 1}), unused) ? "true" : "false");
    return 0;
}
```

## 最小片段二：BBox 与相机 fit

空盒用 `min > max` 构造出来；`radius()` 是半对角线（外接球半径）；fit 公式 `distance = r / sin(halfFov)`，
halfFov 取两个 fov 里更紧的那个。

```bash
g++ -std=c++20 -Wall -Wextra -ffp-contract=off snippet2.cpp -o snippet2 && ./snippet2
```

```cpp
// snippet2.cpp — BBox（空盒 = min > max）+ 包围球 fit：distance = r / sin(halfFov)
// g++ -std=c++20 -Wall -Wextra -ffp-contract=off snippet2.cpp -o snippet2 && ./snippet2
#include <cmath>
#include <cstdio>
#include <initializer_list>

struct Vec3 { float x{}, y{}, z{}; };
inline Vec3 operator+(const Vec3& a, const Vec3& b) { return {a.x + b.x, a.y + b.y, a.z + b.z}; }
inline Vec3 operator-(const Vec3& a, const Vec3& b) { return {a.x - b.x, a.y - b.y, a.z - b.z}; }
inline Vec3 operator*(const Vec3& a, float s) { return {a.x * s, a.y * s, a.z * s}; }
inline float length(const Vec3& a) { return std::sqrt(a.x * a.x + a.y * a.y + a.z * a.z); }

struct BBox {
    Vec3 min{1e30f, 1e30f, 1e30f};    // 初始"空"：min > max，grow 之后才有意义
    Vec3 max{-1e30f, -1e30f, -1e30f};
    void grow(const Vec3& p) {
        min = {p.x < min.x ? p.x : min.x, p.y < min.y ? p.y : min.y, p.z < min.z ? p.z : min.z};
        max = {p.x > max.x ? p.x : max.x, p.y > max.y ? p.y : max.y, p.z > max.z ? p.z : max.z};
    }
    bool empty() const { return min.x > max.x || min.y > max.y || min.z > max.z; }
    Vec3 center() const { return empty() ? Vec3{} : (min + max) * 0.5f; }
    float radius() const { return empty() ? 0.0f : length(max - center()); }   // 半对角线 = 外接球半径
};

int main() {
    const float fovY = 60.0f * 3.14159265f / 180.0f;
    for (float side : {1.6f, 1600.0f}) {
        BBox b;
        b.grow({-side / 2, -side / 2, -side / 2});
        b.grow({side / 2, side / 2, side / 2});
        const float r = b.radius(), aspect = 16.0f / 9.0f;
        const float fovX = 2.0f * std::atan(std::tan(fovY * 0.5f) * aspect);
        const float halfFov = 0.5f * (fovX < fovY ? fovX : fovY);   // 取更紧的那个
        std::printf("side=%-6g r=%.6f distance=%.6f near=%.6f far=%.6f\n",
                    double(side), double(r), double(r / std::sin(halfFov)),
                    double(r / 1000.0f), double(r * 10.0f));
    }
    {   // 竖屏：aspect < 1 时水平 fov 更紧，只用 fovY 会左右裁切
        BBox b;
        b.grow({-0.8f, -0.8f, -0.8f});
        b.grow({0.8f, 0.8f, 0.8f});
        const float r = b.radius(), aspect = 9.0f / 16.0f;
        const float fovX = 2.0f * std::atan(std::tan(fovY * 0.5f) * aspect);
        const float halfFov = 0.5f * (fovX < fovY ? fovX : fovY);
        std::printf("portrait aspect=%.4f: fovX=%.4f rad, distance=%.6f\n",
                    double(aspect), double(fovX), double(r / std::sin(halfFov)));
    }
    const BBox e;   // 空盒：center/radius 返回零，相机 fit 消费它们时不会拿到垃圾
    const Vec3 c = e.center();
    std::printf("empty box: center=(%g,%g,%g) radius=%g\n",
                double(c.x), double(c.y), double(c.z), double(e.radius()));
    return 0;
}
```

实跑输出：

```text
side=1.6    r=1.385641 distance=2.771281 near=0.001386 far=13.856406
side=1600   r=1385.640625 distance=2771.281250 near=1.385641 far=13856.406250
portrait aspect=0.5625: fovX=0.6280 rad, distance=4.486028
empty box: center=(0,0,0) radius=0
```

1.6 的立方体与 1600 的立方体，distance/near/far 恰好差 1000 倍——fit 存在的理由就是"模型单位是 mm 还是 m 都立刻可见"。
竖屏那行 distance 变大，是因为水平 fov 更紧、要退得更远。

## 指回本仓库：生产版本多了什么、为什么多

片段是骨架。生产版在 `src/math/`（五个文件全是 .h，没有 .cpp），逐条说"多了什么、为什么"。

### vec.h：自由函数、零向量约定、header-only

- **自由函数而不是成员**（`vec.h:9-11`）：`dot(a,b)`/`cross(a,b)` 读起来对称；`a.dot(b)` 会诱导 `a.cross(b).dot(c)` 这种链式表达式，在推导阶段（法线、边函数、平面方程）反而更容易看错。
- **没有 SIMD 类型**（`vec.h:13-15`）：本项目的向量化发生在光栅 inner loop（M9，一次 8 个像素），不是"一次 3 个分量"。`float[3]` 结构体更容易验证正确性；注释的声称是 Release 下编译器本来就能自动向量化这些短小函数。
- **零向量约定**（`vec.h:64-74`）：阈值写成 `!(l2 > 1e-16f)`——NaN 与任何数比较都是 false，所以 NaN 也走安全分支。这不是防御式编程的洁癖：真实 OBJ 里重合顶点、零面积面一定会出现，NaN 一旦进光照就污染整块区域，症状"某片区域偶发变黑"，定位成本远高于这一行 `if`。约定由 `test_vec.cpp:48-59`（`normalize_of_zero_vector_is_zero_not_nan`）钉住，连 1e-12 的极短向量也测。
- **C++ 侧**：函数全 `inline` 在头里，因为每个 TU 都要包含，`inline` 允许跨 TU 多重定义，短函数不值得一个 .cpp。`float x{}`（NSDMI）配聚合初始化 `Vec3{1,2,3}`，让 `Mat4 r{}` 全零。一个解析坑：`Vec2 s{1,2} + ...` 会被解析成声明（`test_vec.cpp:89`）。
- **为什么这里没有 `static_assert(sizeof...)` 而 `vertex.h:23-24` 有**：`Varyings` 要被平面方程按 12 个 float 建数组、被 SIMD 一次 4 个分量地读，布局是接口的一部分；`Vec3` 从不被重解释，`sizeof` 不是契约。

### mat.h：不可交换与相对阈值

- 行主序 + 列向量右乘写在 `mat.h:9-14`。不可交换**必须由测试守住**（`test_mat.cpp:45-67`），注释点名的症状就是"所有物体按缩放系数漂移"——片段一的 `2 3 4` vs `4 9 16` 是它的算术。
- **相对奇异阈值**（`mat.h:161-169`）：`eps = 1e-8·maxAbs³`。大坐标端确实单位无关：实测 `scale(2,0.5,4)` 与 `scale(2000,500,4000)`，det 从 4 到 4e9、eps 从 6.4e-7 到 640，det/eps 都是 6.25e6，都可逆。**但小坐标端不是**：`upper3x3AsMat4()` 也置 `m[3][3]=1`（`mat.h:110-116`），maxAbs 至少是 1，阈值退化回固定 1e-8——悬崖在 s³ = 1e-8 即 s ≈ 2.154e-3，实测 scale(3e-3) 可逆、scale(1e-3) 被判奇异，makeNormalMatrix 同在这条悬崖下。这是一条能力边界，不是 bug 叙述；两条边界都由 `test_mat.cpp` 的 `mat4_tryInvert_rejects_singular_and_leaves_output_untouched` 钉住。
- **两种变异，两个判别器**——夹具对它有盲区就抓不到它：变异"完全没阈值（det != 0）"被 `scale(1e-5,1e-5,1)` 抓，det = 1e-10 ≠ 0 但条件数 1e5 会把法线放大五个数量级（Ruling 88；`test_transform.cpp:219-224`）。变异"阈值写成固定 1e-8"**它抓不到**：1e-10 < 1e-8，两种阈值同判奇异，夹具对该错误对称；能抓它的只有大坐标端压扁阵 `scale(1e3,1e3,1e-6)`——det = 1.0 > 1e-8 固定阈值放行，而相对 eps = 1e-8·1e9 = 10 拒绝（条件数 1e9，数学上可逆、数值上不该求逆）。
- **失败 ⇒ false 且不写 out**（`mat.h:26-28`）：`test_mat.cpp:120-134` 用容差 0 钉住"一个元素都不动"。失败时若写 NaN，调用方会静默拿坏矩阵继续画，症状是"某个模型偶尔变形"而不是"这次构建失败"。
- **先取 3x3 再求逆**（`transform.h:62-64`）：相对阈值按整个 4x4 的量级算，大平移列会污染 maxAbs。实测：`tryInvert(translation(1000,2000,3000))` 返回 false（det=1 但 eps=1e-8·3000³=270），而它的 upper3x3 可逆。这就是 `makeNormalMatrix` 先降 3x3 的原因。

### transform.h：决定下游的两条约定与退化兜底

- 头注释（`transform.h:9-13`）就是约定 2、3：视图空间朝 -z；`makePerspective` 产出 clip.z ∈ [-w,w]。**换任何一条都牵动裁剪的六个平面、深度比较的方向、背面剔除的绕向**——注释原话，不是修辞。
- `m[3][2] = -1`（`transform.h:28`）使 w' = -z，是**透视除法的全部来源**；配 z' 那一行（25-27），近平面映到 NDC -1、远平面 +1（`test_transform.cpp:66-80`）。
- **退化输入 ⇒ 单位阵**（`transform.h:16-19`）：fov/aspect/near 非正、far ≤ near 都返回 identity，`!(x > 0)` 的写法顺带挡 NaN（实测 NaN 也触发守卫）。为什么 identity 是诚实的兜底：NaN 进投影矩阵 ⇒ 整帧静默消失；identity ⇒ "什么都没变"，后者好定位得多。同一条规则在 `fitCamera`：bounds 空或 r ≈ 0 ⇒ 原样返回 base，不做"小正数 distance"替代（`scene/fit.h:50-56`，Ruling 49：退化输入的安全响应是**不动作**）。
- **makeLookAt 的两支守卫**（`transform.h:36-43`）：eye == target ⇒ z 退回 {0,0,1}（37 行）；up 与 z 平行 ⇒ 换一个与 z 最不共线的轴（39-43 行：|z.y| > 0.9 取 x 轴，否则 y 轴）。`test_transform.cpp:56-64` 钉住"退化时仍有限、eye 仍在原点"。NaN 进 view 的症状是整个场景消失且没有任何报错（`transform.h:34`）。

### 法线矩阵：inverse-transpose 是几何必然，不是仪式

法线的定义性质是**垂直于表面切向**，不是"长度为 1"。切向 t 满足 n·t = 0；变换后切向是 Mt，要求 n'·(Mt) = 0 对所有 t 成立
⇒ n' = (M⁻¹)ᵀn，因为 ((M⁻¹)ᵀn)ᵀ(Mt) = nᵀM⁻¹Mt = nᵀt = 0（`transform.h:55-60`，M6 笔记 §②）。均匀缩放下它退化成
"归一化一下就行"，所以仪式看起来多余——**非均匀缩放才是判别器**。片段一的数就是这个几何：scale(2,1,1) 下 n=(1,1,0)/√2、
单位切向 t1=(1,-1,0)/√2，朴素变换点积 = **1.5**，垂直性丢了；逆转置 diag(0.5,1,1) 把点积恢复到 0。仓库测试用未归一化切向
(1,-1,0)，所以那边的数是 3/√2 ≈ **2.1213**（`test_transform.cpp:175-177`）——同一个几何，切向长度不同。连带两条，各有自己的变异：

- **归一化在变换之后**（D4，`test_transform.cpp:187-207`）：输入 (1,1,0) 长 √2，经 diag(0.5,1,1) 长 √1.25 ≈ 1.118；只在乘法前归一化，输出长 √0.625 ≈ 0.7906 而不是 1（M6 笔记 §②的实测红值）。
- **nm 哨兵初始化成 model**（`test_transform.cpp:179-181`）：makeNormalMatrix 若没写 out（桩/失败），红会落在几何断言上而不是空转。

### bbox.h / scene/fit.h：空盒与包围球

- 空盒是**构造出来**的 min > max（`bbox.h:10-14`，±1e30 哨兵），不是额外 bool；`center()`/`radius()` 对空盒返回零，因为**相机 fit 会直接消费它们**（`bbox.h:11,27-28`）——返回垃圾值，相机就飞到 1e30 去了。
- `radius()` = 半对角线 = **外接球半径**：8 个角都在球上或球内（`scene/fit.h:30-32`；单位立方体半径 √3，`test_bbox.cpp:20-26`）。
- `distance = r/sin(fov/2)` 用 sin 是因为**界是球**：半径 r 的球在距离 d 处轮廓半角是 asin(r/d)；tan 对应"平面张角"，用在球上差一个 1/cos，把盒子裁掉一圈（`scene/fit.h:29-32`）。
- **取 min(fovX, fovY)**：竖屏（aspect < 1）水平 fov 更紧，只用 fovY 会左右裁切（`scene/fit.h:34`）。实测同一个 1.6 立方体：横屏 distance 2.771281，竖屏 4.486028。
- near = r/1000、far = r·10（`scene/fit.h:63,68`）随模型量级缩放。已知限制：fov < 约 11.478° 时 distance 会超过 far（`scene/fit.h:64-67`），今天 fov 恒 60° 不可达，不加 clamp。

### rect.h / Viewport：y 翻转这一句话

`Rect` 是像素闭区间，空用 x1 < x0 表示，没有"empty() 为假但区间反了"的第三种状态（`rect.h:7-14`）；`clampedTo` 夹完可能变空，
调用方必须检查（`rect.h:16-24`）。Viewport 在 `src/render/viewport.h`（不在 rect.h）：**y 翻转发生在这里且只发生在这里**
（`viewport.h:10-12`、`viewport.cpp:13-14`），depth = ndcZ·0.5+0.5 也在这里（`viewport.cpp:15`）——约定 3 的 [0,1] 落点。
这句话的下游：y 翻转只翻有向面积的符号（NDC 正 ⇒ 屏幕 area2 负），肉眼旋向不变——正面肉眼仍是逆时针，但满足 **area2 < 0**。
所以 mesh 生成器统一"从外面看逆时针"（`mesh.cpp:11-13`），剔除状态 `frontFaceIsClockwise = true`（`pipeline.h:107-111`）实际读作"area2 > 0 是背面"（`triangle_setup.cpp:49`）。第 03 章的剔除测试就建在这个符号上。

### 数值纪律（设计 §5 C1′）

单元测试数值断言一律**容差比较**（默认 1e-5 量级），文档不再出现"精确等于"；GCC 构建加 `-ffp-contract=off`——默认 `fast`
会把插值融合成 FMA 而 MSVC 不融合；三角/pow/log2 直接用标准库，**不自建三角函数**：libstdc++ 与 MSVC CRT 不保证逐位一致，
为 bit-stability 自建是计划外的一大块，与学习目标无关。golden 回归只在 MSVC 主路径判定。这就是本章编译命令带 `-ffp-contract=off` 的原因。

## 坑（一手来自 docs/notes/）

**1. 求逆"快写法"是抄来的索引表，本任务已经踩过一次。** `mat.h:146-151` 记着：24 项展开抄错一个下标，症状是"逆矩阵大致对
但某些元素偏一点"，测试里 a·a⁻¹ 离单位阵差 0.2~0.9——不是干脆的崩溃，你会先怀疑测试容差。求逆每帧最多调用几次（法线矩阵、
相机 fit），这里要的是**能核对**，不是快：余子式写法每一项都能对着定义检查。

**2. 绕向：三次返工，全部源于同一个符号（M2 笔记 §3）。** 测试第一版断言"(0,0)-(10,0)-(0,10) 是顺时针 ⇒ 所以它是正面"：前半句对
（顶左→顶右→底左，视觉确实 CW，cross = +100 > 0，`plane_eq.h:32-33`），后半句错——它是**背面**。错不在口算，在**从旋向词推出正面/背面**。
真正的四步链：mesh 从外面看 CCW → view/proj 保持手性进 NDC（shoelace 为正）→ Viewport 翻 y（C8）⇒ **只翻有向面积的符号，不翻肉眼旋向**
⇒ 屏幕空间 **area2 < 0 是正面**，正面肉眼仍是逆时针。"正面视觉上是 CW"这个错词曾住在仓库六处注释里，`178e9db` 用生产几何实测更正——连仓库注释都会错，
所以别信词，信 area2 的符号与语义判据 `dot(外法线, eye - 顶点) > 0`（符号约定改了后者仍成立）。EdgeFunc 的内侧法线只对一种绕向成立，修复是加 sign 参数由 area2 吸收，**不改顶点顺序**——改顺序会让 flat 取值的"第一个顶点"悄悄换掉。

**3. 法线夹具曾经自相矛盾，是实现者抓出来的（Ruling 87）。** 计划第一版写"面由 (1,0,0)、(0,1,0) 张成，法线 (1,1,0)/√2"——
xy 平面的法线是 (0,0,1)，(1,1,0)/√2 躺在面内；且 z 方向法线在 scale(2,1,1) 下不变，"朴素变换失去垂直性"在那个夹具上
**不可能出现**，测试根本写不成。实现者按设计的 n 与 scale 反推出正确的面，并加了前提断言（先证 dot(n,t1) ≈ 0，再证变换后
不垂直，`test_transform.cpp:170-171`）。没有前提断言，夹具写错时测试仍可能"绿"，而绿的含义是"证明了一件没发生的事"。
规则写进了设计：**规格里出现"表面/法线"这类成对对象时，两个都必须写死；只写一个等于把矛盾留给下游发现。**
这是本项目文档第八次被下游抓错。
